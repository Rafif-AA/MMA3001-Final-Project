"""Summarize the loaded YOLO model and measure its activations.Run the evaluation cell first to load the saved model. This cell definesnetwork and creates its own activation table; no earlier profiling cell isrequired. Counts describe the current loaded model, which may be fused orfrozen after evaluation. Gradient flags do not indicate training history."""import pandas as pdimport torchfrom IPython.display import displaydef collect_layer_activations(network, image_size=640):    """Return tensor output shapes and activation counts for leaf modules.    Run one dummy image on the model's current device and dtype without    gradients. Always remove hooks and restore module training modes,    including if the forward pass fails. Counts include intermediate leaf    outputs and are not a measurement of GPU memory traffic.    """    rows = []    handles = []    modes = [(module, module.training) for module in network.modules()]    reference = next(network.parameters())    def make_hook(name):        """Create a hook that records tensor outputs for a named module."""        def hook(module, inputs, output):            """Record output shape and per-image tensor activation count."""            if isinstance(output, torch.Tensor):                rows.append({                    "Layer": name,                    "Type": type(module).__name__,                    "Output Shape": tuple(output.shape),                    "Activations": output.numel(),  # Dummy batch size is one.                })        return hook    try:        network.eval()        for name, module in network.named_modules():            if name and not list(module.children()):                handles.append(module.register_forward_hook(make_hook(name)))        dummy_input = torch.zeros(            (1, 3, image_size, image_size),            device=reference.device,            dtype=reference.dtype,        )        with torch.no_grad():            network(dummy_input)    finally:        for handle in handles:            handle.remove()        for module, was_training in modes:            module.training = was_training    return pd.DataFrame(rows, columns=[        "Layer", "Type", "Output Shape", "Activations"    ])if "model" not in globals():    raise RuntimeError("Run the evaluation cell first to load the saved YOLO model.")network = model.modeldf = collect_layer_activations(network, image_size=640)total_params = sum(parameter.numel() for parameter in network.parameters())tuned_params = sum(    parameter.numel() for parameter in network.parameters()    if parameter.requires_grad)total_layers = sum(1 for _ in network.modules())total_neurons = sum(    getattr(module, "out_channels", getattr(module, "out_features", 0))    for module in network.modules())total_activations = int(df["Activations"].sum())display(df)print("=" * 62)print("YOLO26n LOADED MODEL SUMMARY")print("=" * 62)for label, value in [    ("Modules (including containers)", total_layers),    ("Output channels/features (sum)", total_neurons),    ("Recorded activations (640x640)", total_activations),    ("Total parameters", total_params),    ("Parameters requiring gradients", tuned_params),]:    print(f"{label + ':':<42} {value:,}")print("=" * 62)
